const tryAddColumn = async (conn, table, column, definition) => {
  try {
    await conn.query(`ALTER TABLE \`${table}\` ADD COLUMN \`${column}\` ${definition}`);
    console.log(`   [MIGRATE] + ${table}.${column}`);
  } catch (e) {
    if (e.code !== 'ER_DUP_FIELDNAME' && e.code !== 'ER_NO_SUCH_TABLE') {
      console.warn(`   [MIGRATE] Bỏ qua ${table}.${column}: ${e.message}`);
    }
  }
};

const Payroll = {
  calculateKPI: async (userId, period) => {
    const now = new Date();
    let startDate;
    if (period === 'day') { startDate = new Date(now.setHours(0,0,0,0)); }
    else if (period === 'week') { const day = now.getDay() || 7; startDate = new Date(now.setDate(now.getDate() - day + 1)); }
    else if (period === 'month') { startDate = new Date(now.getFullYear(), now.getMonth(), 1); }
    else if (period === 'quarter') { const q = Math.floor(now.getMonth()/3); startDate = new Date(now.getFullYear(), q*3, 1); }
    else { startDate = new Date(now.getFullYear(), 0, 1); }
    const startStr = startDate.toISOString().slice(0, 19).replace('T', ' ');

    const [rows] = await pool.query(`
      SELECT SUM(kpi_weight) as total_weight, SUM(CASE WHEN status='done' AND evidence IS NOT NULL THEN kpi_weight ELSE 0 END) as done_weight
      FROM tasks WHERE assignee_id = ? AND task_type = 'kpi' AND done_at >= ?
    `, [userId, startStr]);
    
    if (!rows[0].total_weight || rows[0].total_weight == 0) return { kpi: 0, bonus: 0, rank: 'Không có KPI' };
    
    const kpiPercent = (rows[0].done_weight / rows[0].total_weight) * 100;
    let bonus = 0; let rank = 'Chưa đạt';
    if (kpiPercent >= 90) { bonus = 1500000; rank = 'Tốt'; }
    else if (kpiPercent >= 50) { bonus = 700000; rank = 'Trung bình'; }
    return { kpi: kpiPercent, bonus, rank };
  },
  generate: async (userId, period) => {
    const [users] = await pool.query(`SELECT * FROM users WHERE id = ? AND status = 'active'`, [userId]);
    if (users.length === 0) throw new Error('User không tồn tại hoặc đã nghỉ việc');
    const user = users[0];

    const [att] = await pool.query('SELECT SUM(work_hours) as wh, SUM(ot_hours) as oh FROM attendance WHERE user_id = ? AND date LIKE ?', [userId, period + '%']);
    const workHours = Number(att[0].wh || 0);
    const otHours = Number(att[0].oh || 0);

    const hourlyRate = user.salary_base / (26 * 8);
    const workPay = workHours * hourlyRate;
    const otPay = otHours * hourlyRate * 1.5;

    const kpiData = await Payroll.calculateKPI(userId, 'month');
    const bhxh = user.salary_base * 0.105;
    const net = workPay + otPay + kpiData.bonus - bhxh;

    await pool.query(`
      INSERT INTO payrolls (period, user_id, salary_base, work_pay, ot_pay, kpi_bonus, bhxh_deduction, net_salary, status)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'draft')
      ON DUPLICATE KEY UPDATE work_pay = VALUES(work_pay), ot_pay = VALUES(ot_pay), kpi_bonus = VALUES(kpi_bonus), net_salary = VALUES(net_salary)
    `, [period, userId, user.salary_base, workPay, otPay, kpiData.bonus, bhxh, net]);

    return { kpi: kpiData, workPay, otPay, net };
  }
};

function positiveId(value, field) {

function money(value, field, fallback = 0) {
  if (value === undefined || value === null || value === '') {
    return fallback;
  }

  const n = Number(value);
  if (!Number.isSafeInteger(n) || n < 0) {
    throw badRequest(`${field} pháº£i lÃ  sá»‘ nguyÃªn khÃ´ng Ã¢m`);

async function recalcTotal(orderId, conn = null) {
  const id = positiveId(orderId, 'orderId');
  const exec = executor(conn);

  // Khi khÃ´ng cÃ³ transaction bÃªn ngoÃ i,
  // tá»± táº¡o transaction Ä‘á»ƒ trÃ¡nh cáº­p nháº­t ná»­a chá»«ng.
  const ownConn = !conn;
  const db = ownConn ? await pool.getConnection() : exec;

  try {
    if (ownConn) await db.beginTransaction();

    const [orders] = await db.query(`
      SELECT discount, tax_amount, shipping_fee,
             sub_total, paid_amount
      FROM orders
      WHERE id = ?
      FOR UPDATE
    `, [id]);

    if (!orders.length) {
      throw badRequest(`KhÃ´ng tÃ¬m tháº¥y Ä‘Æ¡n id=${id}`);
    }

    const order = orders[0];

    const [sums] = await db.query(`
      SELECT COALESCE(SUM(amount), 0) AS subtotal
      FROM order_items
      WHERE order_id = ?
    `, [id]);

    const subTotal = money(sums[0].subtotal, 'sub_total');
    const discount = money(order.discount, 'discount');
    const shipping = money(order.shipping_fee, 'shipping_fee');
    const previousTax = money(order.tax_amount, 'tax_amount');
    const previousSubTotal = money(order.sub_total, 'sub_total');

    // Schema chÆ°a cÃ³ tax_mode/tax_rate.
    // Quy Æ°á»›c tÆ°Æ¡ng thÃ­ch táº¡m thá»i:
    // - Thuáº¿ 0: tá»± tÃ­nh VAT máº·c Ä‘á»‹nh.
    // - Thuáº¿ cÅ© báº±ng VAT máº·c Ä‘á»‹nh cá»§a subtotal cÅ©:
    //   tiáº¿p tá»¥c tÃ­nh tá»± Ä‘á»™ng.
    // - Thuáº¿ khÃ¡c: giá»¯ nguyÃªn nhÆ° giÃ¡ trá»‹ nháº­p tay.
    //
    // LÆ°u Ã½: quy Æ°á»›c nÃ y khÃ´ng phÃ¢n biá»‡t Ä‘Æ°á»£c
    // VAT 0% chá»§ Ä‘á»™ng vá»›i VAT chÆ°a Ä‘Æ°á»£c nháº­p.
    const oldAutoTax = Math.round(
      previousSubTotal * DEFAULT_TAX_RATE
    );

    let taxAmount = previousTax;

    if (previousTax === 0 || previousTax === oldAutoTax) {
      taxAmount = Math.round(
        subTotal * DEFAULT_TAX_RATE
      );
    }

    if (discount > subTotal) {
      throw badRequest(
        'Giáº£m giÃ¡ khÃ´ng Ä‘Æ°á»£c lá»›n hÆ¡n tiá»n hÃ ng'
      );
    }

    const total =
      subTotal - discount + taxAmount + shipping;

    if (!Number.isSafeInteger(total) || total < 0) {
      throw badRequest('Tá»•ng tiá»n khÃ´ng há»£p lá»‡');
    }

    const paidAmount = money(
      order.paid_amount, 'paid_amount'
    );

    // KhÃ´ng tá»± giáº£m tiá»n Ä‘Ã£ thu.
    // Náº¿u sá»­a Ä‘Æ¡n khiáº¿n tiá»n Ä‘Ã£ thu vÆ°á»£t tá»•ng,
    // pháº£i xá»­ lÃ½ hoÃ n tiá»n riÃªng.
    if (paidAmount > total) {
      throw badRequest(
        'Tá»•ng tiá»n má»›i nhá» hÆ¡n sá»‘ tiá»n Ä‘Ã£ thanh toÃ¡n'
      );
    }

    const paymentStatus =
      paidAmount === 0 ? 'unpaid' :
      paidAmount >= total ? 'paid' : 'partial';

    await db.query(`
      UPDATE orders
      SET sub_total = ?,
          tax_amount = ?,
          total = ?,
          payment_status = ?
      WHERE id = ?
    `, [
      subTotal,
      taxAmount,
      total,
      paymentStatus,
      id
    ]);

    if (ownConn) await db.commit();

    return {
      sub_total: subTotal,
      discount,
      tax_amount: taxAmount,
      shipping_fee: shipping,
      total,
      paid_amount: paidAmount,

    };
  } catch (err) {

    throw err;
  } finally {

  }
}

async function withTransaction(conn, callback) {
  if (conn) {
    return callback(conn);
  }

  const db = await pool.getConnection();

  try {
    await db.beginTransaction();

    const result = await callback(db);

    await db.commit();
    return result;
  } catch (err) {
    try {
      await db.rollback();
    } catch (rollbackErr) {
      console.error(
        '[OrderItem] Rollback error:',
        rollbackErr.message
      );
    }

    throw err;
  } finally {
    db.release();
  }
}

        orderId,

        data.item_id !== undefined
          ? validId(data.item_id, 'item_id')

      const qty =
        data.qty !== undefined
          ? validQty(data.qty)
          : validQty(existing.qty);

        qty,
        unitPrice

        SET item_id = ?,
            qty = ?,
            unit_price = ?,
            amount = ?
        WHERE id = ?
      `, [
        itemId,
        qty,
        unitPrice,
        amount,
        rowId
      ]);

      await Order.recalcTotal(
        orderId,
        db
      );

      return result.affectedRows > 0;
    });
  }

const express = require('express');
const router = express.Router();
const jwt = require('jsonwebtoken');
const User = require('../models/User');
const { comparePassword } = require('../utils/password');
const auth = require('../middleware/auth');

router.post('/login', async (req, res) => {
  try {
    const { username, password } = req.body;
    const user = await User.findByUsername(username);
    if (!user) return res.status(401).json({ error: 'Sai tài khoản hoặc mật khẩu' });

    const isMatch = await comparePassword(password, user.password_hash);
    if (!isMatch) return res.status(401).json({ error: 'Sai tài khoản hoặc mật khẩu' });

    const token = jwt.sign(
      { id: user.id, role: user.role, team_id: user.team_id, name: user.name },
      process.env.JWT_SECRET,
      { expiresIn: process.env.JWT_EXPIRES_IN }
    );
    res.json({ token, role: user.role, name: user.name });
  } catch (err) { res.status(500).json({ error: err.message }); }
});

router.get('/me', auth, async (req, res) => {
  try {
    const user = await User.findById(req.user.id);
    res.json(user);
  } catch (err) { res.status(500).json({ error: err.message }); }
});

module.exports = router;

const express = require('express');
const router = express.Router();
const Item = require('../models/Item');
const auth = require('../middleware/auth');
const { allow } = require('../middleware/rbac');
const { logAudit } = require('../middleware/audit');
const { generateNextCode } = require('../utils/codeGenerator');
const { pool } = require('../config/db');

// ============================================================
// HELPER: Validate dữ liệu đầu vào
// ============================================================
const VALID_TYPES = ['material', 'product'];

const validateItemInput = (body, isUpdate = false) => {
  const errors = [];

  // type: bắt buộc khi tạo mới
  if (!isUpdate || body.type !== undefined) {
    if (!body.type) {
      errors.push('Thiếu trường "type" (material hoặc product)');
    } else if (!VALID_TYPES.includes(body.type)) {
      errors.push(`"type" phải là "material" hoặc "product" (nhận được: "${body.type}")`);
    }
  }

  // name: bắt buộc khi tạo mới
  if (!isUpdate || body.name !== undefined) {
    if (!body.name || String(body.name).trim() === '') {
      errors.push('Thiếu trường "name" (tên hàng hóa không được rỗng)');
    } else if (String(body.name).length > 255) {
      errors.push('"name" không được vượt quá 255 ký tự');
    }
  }

  // unit: bắt buộc khi tạo mới
  if (!isUpdate || body.unit !== undefined) {
    if (!body.unit || String(body.unit).trim() === '') {
      errors.push('Thiếu trường "unit" (đơn vị tính)');
    } else if (String(body.unit).length > 20) {
      errors.push('"unit" không được vượt quá 20 ký tự');
    }
  }

  // code: nếu user tự truyền thì phải đúng format
  if (body.code !== undefined && body.code !== null && String(body.code).trim() !== '') {
    if (!/^[A-Za-z0-9_-]{1,50}$/.test(String(body.code))) {
      errors.push('"code" chỉ được chứa chữ, số, gạch ngang, gạch dưới (tối đa 50 ký tự)');
    }
  }

  // Các trường số không được âm
  const numericFields = ['price_in', 'price_out', 'price_floor', 'min_stock', 'max_stock', 'weight'];
  for (const f of numericFields) {
    if (body[f] !== undefined && body[f] !== null && body[f] !== '' && Number(body[f]) < 0) {
      errors.push(`"${f}" không được âm`);
    }
  }

  // max_stock phải >= min_stock (nếu cả 2 cùng được gửi)
  if (body.min_stock !== undefined && body.max_stock !== undefined) {
    if (Number(body.max_stock) > 0 && Number(body.max_stock) < Number(body.min_stock)) {
      errors.push('"max_stock" phải lớn hơn hoặc bằng "min_stock"');
    }
  }

  return errors;
};

// ============================================================
// GET /api/items — Lấy danh sách (lọc ?type=material hoặc ?type=product)
// ============================================================
router.get('/', auth, allow('items'), async (req, res) => {
  const type = req.query.type;
  if (type && !VALID_TYPES.includes(type)) {
    return res.status(400).json({
      success: false,
      error: `"type" phải là "material" hoặc "product" (nhận được: "${type}")`
    });
  }
  const data = await Item.getAll(type);
  res.json({ success: true, data, count: data.length });
});

// ============================================================
// POST /api/items — Tạo hàng hóa mới (có RETRY LOOP chống trùng mã)
// ============================================================
router.post('/', auth, allow('items'), async (req, res) => {
  // 1. Validate đầu vào
  const errors = validateItemInput(req.body, false);
  if (errors.length > 0) {
    return res.status(400).json({ success: false, error: errors.join('; '), errors });
  }

  const body = { ...req.body };
  const prefix = body.type === 'material' ? 'VT' : 'TP';
  const whereClause = "AND type = '" + body.type + "'";

  // 2. RETRY LOOP: nếu gặp race condition (ER_DUP_ENTRY), tự tăng mã và thử lại
  let attempts = 0;
  const MAX_ATTEMPTS = 5;
  let lastError = null;

  while (attempts < MAX_ATTEMPTS) {
    attempts++;

    // 2.1 Sinh mã nếu user không tự cung cấp
    if (!body.code || String(body.code).trim() === '') {
      body.code = await generateNextCode('items', 'code', prefix, 3, whereClause);
    }

    try {
      // 2.2 Tạo hàng hóa
      const id = await Item.create(body);

      // 2.3 Đọc lại object đầy đủ
      const [createdRows] = await pool.query('SELECT * FROM items WHERE id = ?', [id]);
      const created = createdRows[0];

      // 2.4 Audit
      await logAudit(
        req.user.id,
        'CREATE',
        'items',
        id,
        null,
        { code: body.code, name: body.name, type: body.type },
        null,
        req.ip
      );

      // 2.5 Trả về FULL object
      return res.status(201).json({
        success: true,
        id,
        code: body.code,
        data: created,
        attempt: attempts
      });
    } catch (err) {
      lastError = err;

      // Nếu trùng mã → reset và thử lại
      if (err.code === 'ER_DUP_ENTRY' && err.message.includes('code')) {
        console.warn(
          `[POST /items] Trùng mã "${body.code}" (lần ${attempts}/${MAX_ATTEMPTS}), ` +
          `đang tự tăng và thử lại...`
        );
        body.code = null;
        continue;
      }

      order.sub_total =
        totals.sub_total;

      order.total =
        totals.total;

      // --------------------------------------------------------
      // FULL PAYMENT
      // --------------------------------------------------------

      const paidAmount =
        Number(order.paid_amount || 0);

      const total =
        Number(order.total || 0);

      const requireFullPayment =
        Boolean(
          Number(
            order.require_full_payment
          )
        );

      if (
        requireFullPayment &&
        paidAmount < total
      ) {
        throw makeError(

          `ÄÃ£ thu ${paidAmount.toLocaleString('vi-VN')} / ` +

      // --------------------------------------------------------
      // CREDIT LIMIT
      // Chá»‰ SALE
      // --------------------------------------------------------

      if (order.type === 'sale') {
        const [partners] =
          await conn.query(
            `
              SELECT
                id,

// HELPER: Validate dữ liệu đầu vào
// ============================================================
const VALID_TYPES = ['customer', 'supplier'];

const validatePartnerInput = (body, isUpdate = false) => {
  const errors = [];

  // type: bắt buộc khi tạo mới, nếu có khi update thì phải hợp lệ
  if (!isUpdate || body.type !== undefined) {
    if (!body.type) {
      errors.push('Thiếu trường "type" (customer hoặc supplier)');
    } else if (!VALID_TYPES.includes(body.type)) {
      errors.push(`"type" phải là "customer" hoặc "supplier" (nhận được: "${body.type}")`);
    }
  }

  // name: bắt buộc khi tạo mới, nếu có khi update thì không được rỗng
  if (!isUpdate || body.name !== undefined) {
    if (!body.name || String(body.name).trim() === '') {
      errors.push('Thiếu trường "name" (tên đối tác không được rỗng)');
    } else if (String(body.name).length > 255) {
      errors.push('"name" không được vượt quá 255 ký tự');
    }
  }

  // code: nếu user tự truyền thì phải đúng format (không chứa ký tự lạ)
  if (body.code !== undefined && body.code !== null && String(body.code).trim() !== '') {
    if (!/^[A-Za-z0-9_-]{1,50}$/.test(String(body.code))) {
      errors.push('"code" chỉ được chứa chữ, số, gạch ngang, gạch dưới (tối đa 50 ký tự)');
    }
  }

  // credit_limit, payment_term: nếu có thì phải >= 0
  if (body.credit_limit !== undefined && Number(body.credit_limit) < 0) {
    errors.push('"credit_limit" không được âm');
  }
  if (body.payment_term !== undefined && Number(body.payment_term) < 0) {
    errors.push('"payment_term" không được âm');
  }

  return errors;
};

// POST /api/partners — Tạo đối tác mới (có RETRY LOOP chống trùng mã)
// ============================================================
router.post('/', auth, allow('partners'), async (req, res) => {
  // 1. Validate đầu vào
  const errors = validatePartnerInput(req.body, false);
  if (errors.length > 0) {
    return res.status(400).json({ success: false, error: errors.join('; '), errors });
  }

  const body = { ...req.body };
  const prefix = body.type === 'supplier' ? 'NCC' : 'KH';
  const whereClause = "AND type = '" + body.type + "'";

  // 2. RETRY LOOP: nếu gặp race condition (ER_DUP_ENTRY), tự tăng mã và thử lại
  let attempts = 0;
  const MAX_ATTEMPTS = 5;
  let lastError = null;

  while (attempts < MAX_ATTEMPTS) {
    attempts++;

    // 2.1 Sinh mã nếu user không tự cung cấp
    if (!body.code || String(body.code).trim() === '') {
      body.code = await generateNextCode('partners', 'code', prefix, 3, whereClause);
    }

    try {
      // 2.2 Tạo đối tác
      const id = await Partner.create(body);

      // 2.3 Đọc lại toàn bộ object từ DB để trả về FE ngay
      const [createdRows] = await pool.query('SELECT * FROM partners WHERE id = ?', [id]);
      const created = createdRows[0];

      // 2.4 Ghi audit log
      await logAudit(
        req.user.id,
        'CREATE',
        'partners',
        id,
        null,
        { code: body.code, name: body.name, type: body.type },
        null,
        req.ip
      );

      // 2.5 Trả về FULL object
      return res.status(201).json({
        success: true,
        id,
        code: body.code,
        data: created,
        attempt: attempts
      });
    } catch (err) {
      lastError = err;

      // Nếu lỗi là "Duplicate entry" trên cột code → reset mã và thử lại
      if (err.code === 'ER_DUP_ENTRY' && err.message.includes('code')) {
        console.warn(
          `[POST /partners] Trùng mã "${body.code}" (lần ${attempts}/${MAX_ATTEMPTS}), ` +
          `đang tự tăng và thử lại...`
        );
        // Xóa code cũ để lần sau generateNextCode sinh mã mới
        body.code = null;
        continue;
      }

      // Lỗi khác → throw để global handler bắt
      throw err;
    }
  }

  // Hết MAX_ATTEMPTS vẫn trùng
  return res.status(409).json({
    success: false,
    error: `Không thể sinh mã duy nhất sau ${MAX_ATTEMPTS} lần thử. ` +
           `Vui lòng thử lại hoặc kiểm tra bảng partners có dữ liệu lỗi.`,
    lastError: lastError?.message
  });
});

// PATCH /api/partners/:id — Cập nhật
// ============================================================
router.patch('/:id', auth, allow('partners'), async (req, res) => {
  const id = req.params.id;

  // 1. Kiểm tra tồn tại
  const old = await Partner.findById(id);
  if (!old) {
    return res.status(404).json({ success: false, error: 'Không tìm thấy đối tác' });
  }

  // 2. Validate
  const errors = validatePartnerInput(req.body, true);
  if (errors.length > 0) {
    return res.status(400).json({ success: false, error: errors.join('; '), errors });
  }

  // 3. Update
  await Partner.update(id, { ...old, ...req.body });

  // 4. Đọc lại object sau update
  const [updatedRows] = await pool.query('SELECT * FROM partners WHERE id = ?', [id]);

  await logAudit(req.user.id, 'UPDATE', 'partners', id, old, req.body, null, req.ip);

  res.json({
    success: true,
    message: 'Cập nhật thành công',
    data: updatedRows[0]
  });
});

router.delete('/:id', auth, allow('partners'), async (req, res) => {

  const old = await Partner.findById(id);
  if (!old) {
    return res.status(404).json({ success: false, error: 'Không tìm thấy đối tác' });
  }

  if (!old.is_active) {
    return res.status(400).json({ success: false, error: 'Đối tác đã bị khóa trước đó' });
  }

  await pool.query('UPDATE partners SET is_active = FALSE WHERE id = ?', [id]);
  await logAudit(
    req.user.id,
    'DELETE',
    'partners',
    id,
    { is_active: true },
    { is_active: false },
    'Khóa đối tác',
    req.ip
  );

  res.json({ success: true, message: 'Đã khóa đối tác' });
});

router.get('/balance', auth, async (req, res) => {
  try {
    const [rows] = await pool.query(`
      SELECT i.id, i.code, i.name, i.min_stock, COALESCE(SUM(s.signed_qty), 0) as stock 
      FROM items i 
      LEFT JOIN stock_ledger s ON i.id = s.item_id 
      GROUP BY i.id
    `);
    res.json({ success: true, data: rows });
  } catch (err) { res.status(500).json({ error: err.message }); }
});

router.post('/', auth, async (req, res) => {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    const { item_id, type, qty } = req.body;
    if (Number(qty) <= 0) throw new Error('Số lượng phải > 0');

    if (type === 'out') {
      const [stock] = await conn.query('SELECT COALESCE(SUM(signed_qty), 0) as stock FROM stock_ledger WHERE item_id = ?', [item_id]);
      if (Number(stock[0].stock) < Number(qty)) throw new Error(`Không đủ tồn kho để xuất (Hiện có: ${stock[0].stock})`);
    }

    const signed = type === 'out' ? -Math.abs(qty) : Math.abs(qty);
    await StockLedger.create({ ...req.body, signed_qty: signed, created_by: req.user.id });
    await logAudit(req.user.id, 'CREATE', 'stock_ledger', null, null, req.body, null, req.ip);
    await conn.commit();
    res.status(201).json({ success: true });
  } catch (err) { await conn.rollback(); res.status(400).json({ error: err.message }); } finally { conn.release(); }
});

const normalizeStr = (str) => {
  if (!str) return '';
  let res = str.toLowerCase();
  res = res.replace(/[\uFFFD\?]/g, '');
  res = res.normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/đ/g, 'd');
  res = res.replace(/[^a-z0-9]/g, '');
  return res;
};

const buildRevenueContext = async (role, period) => {

const buildKPIContext = async () => {

const buildViolationContext = async (qSafe, question) => {
  let ctx = '';
  let directAnswer = null;
  try {
    // ─── TẦNG 1: Mã VP cụ thể (VP001, VP002...) ───
    const vpMatch = question.match(/VP\d{3}/i);
    if (vpMatch) {
      const [v] = await pool.query(`SELECT * FROM violation_rules WHERE code = ?`, [vpMatch[0].toUpperCase()]);
      if (v.length > 0) {
        const x = v[0];
        ctx += `[VI PHẠM ${x.code}]:\n- Nhóm: ${x.group_name}\n- Hành vi: ${x.behavior}\n- Giá trị: ${Number(x.fine_amount).toLocaleString('vi-VN')} VNĐ\n- Xác minh: ${x.verify_method}\n- Khắc phục: ${x.remedy}\n- Quy định: ${x.regulation}\n`;
        directAnswer = `**${x.code} - ${x.behavior}**\n\n- **Nhóm:** ${x.group_name}\n- **Giá trị quy đổi:** ${Number(x.fine_amount).toLocaleString('vi-VN')} VNĐ\n- **Cách xác minh:** ${x.verify_method}\n- **Hướng khắc phục:** ${x.remedy}\n- **Quy định:** ${x.regulation}\n\n*Lưu ý: Đây là quy định tham khảo, mọi xử lý kỷ luật/trừ lương phải do người có thẩm quyền phê duyệt.*`;
        return { context: ctx, directAnswer };
      }
    }

    // ─── TẦNG 2: Hỏi về phiếu chờ giải trình ───
    if (qSafe.includes('phieu') || qSafe.includes('chogiai') || qSafe.includes('incident') || qSafe.includes('hoso')) {
      const [pendingRows] = await pool.query(`
        SELECT i.*, u.full_name as user_name 
        FROM incidents i 
        JOIN users u ON i.user_id = u.id 
        WHERE i.status = 'pending_explanation' 
        ORDER BY i.created_at DESC LIMIT 10
      `);
      if (pendingRows.length > 0) {
        ctx += `[DANH SÁCH PHIẾU CHỜ GIẢI TRÌNH]:\n`;
        pendingRows.forEach(r => {
          ctx += `- ${r.code}: ${r.user_name} - ${r.violation_codes}\n`;
        });
        directAnswer = `Hiện có **${pendingRows.length} phiếu đang chờ giải trình**:\n\n${pendingRows.map(r => `- **${r.code}**: Nhân viên ${r.user_name} - Vi phạm: ${r.violation_codes}`).join('\n')}`;
      } else {
        ctx += `[PHIẾU CHỜ GIẢI TRÌNH]: Không có phiếu nào đang chờ.\n`;
        directAnswer = `Hiện tại không có phiếu vi phạm nào đang chờ giải trình.`;
      }
      return { context: ctx, directAnswer };
    }

    // ─── TẦNG 3: Hành vi cụ thể (đi muộn, về sớm...) ───
    // ĐÃ LOẠI BỎ TỪ KHÓA 'khong' GÂY NHIỄU
    const behaviorKeywords = ['dimu', 'vesom', 'nghiphep', 'chamcong', 'quenghi', 'vangmat', 'doica'];
    const matchedBehavior = behaviorKeywords.find(k => qSafe.includes(k));
    
    if (matchedBehavior) {
      const [allRules] = await pool.query(`SELECT code, behavior, fine_amount, verify_method FROM violation_rules`);
      const matchedRows = allRules.filter(r => {
        const bNorm = normalizeStr(r.behavior || '');
        return bNorm.includes(matchedBehavior);
      });
      
      if (matchedRows.length > 0) {
        ctx += `[QUY ĐỊNH VI PHẠM]:\n`;
        matchedRows.forEach(r => {
          ctx += `- ${r.code}: ${r.behavior} - ${Number(r.fine_amount).toLocaleString('vi-VN')}đ (Xác minh: ${r.verify_method})\n`;
        });
        directAnswer = `Theo quy định của công ty:\n\n${matchedRows.map(r => `- **${r.code}: ${r.behavior}**\n  - Giá trị quy đổi: ${Number(r.fine_amount).toLocaleString('vi-VN')} VNĐ\n  - Cách xác minh: ${r.verify_method}`).join('\n\n')}\n\n*Lưu ý: Đây là quy định tham khảo, mọi xử lý kỷ luật/trừ lương phải do người có thẩm quyền phê duyệt.*`;
        return { context: ctx, directAnswer };
      }
    }

    // ─── TẦNG 4: Hỏi chung về vi phạm ───
    if (qSafe.includes('vipham') || qSafe.includes('phat') || qSafe.includes('kyluat')) {

      });

      const [pendingCount] = await pool.query(`SELECT COUNT(*) as c FROM incidents WHERE status = 'pending_explanation'`);
      ctx += `[TỔNG PHIẾU CHỜ GIẢI TRÌNH]: ${pendingCount[0].c} phiếu\n`;

      return { context: ctx, directAnswer };
    }
    
  } catch (e) {
    console.error('Lỗi buildViolationContext:', e);
    ctx += `[Lỗi đọc vi phạm]: ${e.message}\n`;
  }
  return { context: ctx, directAnswer };

const buildSOPContext = async (qSafe, question) => {
  let ctx = '';
  try {
    const sopMatch = question.match(/[A-Z]{2,4}-\d{2,3}/i);
    if (sopMatch && !sopMatch[0].toUpperCase().startsWith('VP')) {
      const [s] = await pool.query(`SELECT code, title, steps, tags FROM sop_docs WHERE code = ?`, [sopMatch[0].toUpperCase()]);
      if (s.length > 0) {
        ctx += `[SOP ${s[0].code}]:\n- Tên: ${s[0].title}\n- Phân loại: ${s[0].tags}\n- Chi tiết: ${s[0].steps}\n`;
        return ctx;
      }
    }
    const words = qSafe.split(/\s+/).filter(w => w.length > 3);
    if (words.length >= 2) {
      const phrase = words.slice(0, 3).join(' ');
      const [rows] = await pool.query(`SELECT code, title, steps FROM sop_docs WHERE title LIKE ? LIMIT 3`, [`%${phrase}%`]);
      if (rows.length > 0) {
        ctx += `[SOP LIÊN QUAN]:\n`;
        rows.forEach(r => {
          ctx += `- ${r.code}: ${r.title}\n  ${r.steps.substring(0, 300)}...\n`;
        });
      }
    }
  } catch (e) { ctx += `[Lỗi đọc SOP]: ${e.message}\n`; }
  return ctx;
};

const { pool } = require('../config/db');

const scanAndCreateIncidents = async (dateStr = null) => {
  try {
    const targetDate = dateStr || new Date().toISOString().split('T')[0];
    console.log(`🤖 [BOT] Bắt đầu quét vi phạm ngày: ${targetDate}`);

    const [attendances] = await pool.query(`
      SELECT a.*, u.full_name, u.username 
      FROM attendance a 
      JOIN users u ON a.user_id = u.id 
      WHERE a.date = ?
    `, [targetDate]);

    if (attendances.length === 0) {
      console.log(`🤖 [BOT] Không có dữ liệu chấm công ngày ${targetDate}`);
      return { scanned: 0, created: 0 };
    }

    let createdCount = 0;

    for (const att of attendances) {
      if (!att.check_in) continue;

      const checkInTime = new Date(att.check_in);
      const hour = checkInTime.getHours();
      const minute = checkInTime.getMinutes();

      let lateMinutes = 0;
      if (hour >= 8) {
        lateMinutes = (hour - 8) * 60 + minute;
      }

      if (lateMinutes > 0) {
        let violationCode = null;
        if (lateMinutes <= 15) violationCode = 'VP001';
        else if (lateMinutes <= 30) violationCode = 'VP002';
        else violationCode = 'VP003';

        const [existing] = await pool.query(`
          SELECT id FROM incidents 
          WHERE user_id = ? AND DATE(created_at) = ? 
          AND JSON_CONTAINS(violation_codes, JSON_QUOTE(?))
        `, [att.user_id, targetDate, violationCode]);

        if (existing.length === 0) {
          const code = `INC-AUTO-${Date.now()}-${att.user_id}`;
          const evidence = `[Hệ thống tự động] Nhân viên ${att.full_name} (${att.username}) check-in lúc ${checkInTime.toLocaleTimeString('vi-VN')}, trễ ${lateMinutes} phút so với giờ làm việc (8h00).`;

          await pool.query(`
            INSERT INTO incidents (code, user_id, violation_codes, status, evidence, created_by) 
            VALUES (?, ?, ?, 'pending_explanation', ?, 1)
          `, [code, att.user_id, JSON.stringify([violationCode]), evidence]);

          console.log(`🤖 [BOT] ✅ Đã tạo phiếu cho ${att.full_name}: ${violationCode} (Trễ ${lateMinutes} phút)`);
          createdCount++;
        } else {
          console.log(`🤖 [BOT] ⏭️ Bỏ qua ${att.full_name} (Đã có phiếu ${violationCode})`);
        }
      }
    }

    console.log(`🤖 [BOT] Hoàn tất! Đã quét ${attendances.length} nhân viên, tạo mới ${createdCount} phiếu.`);
    return { scanned: attendances.length, created: createdCount };

  } catch (err) {
    console.error('🤖 [BOT] Lỗi quét vi phạm:', err.message);
    throw err;
  }
};

module.exports = { scanAndCreateIncidents };

const { pool } = require('../config/db');

/**
 * Sinh mã tự động KHÔNG TRÙNG (VD: KH001, NCC002, DH00001).
 * 
 * ═══════════════════════════════════════════════════════════
 * CẢI TIẾN so với phiên bản cũ:
 * ───────────────────────────────────────────────────────────
 *  1. Dùng MAX(CAST(SUBSTRING(...))) thay vì ORDER BY id DESC LIMIT 1
 *     → Nếu dòng cuối bị xóa, vẫn sinh đúng số tiếp (KHÔNG trùng).
 *  2. Có vòng lặp retry nội bộ: nếu sinh ra mã đã tồn tại, tự tăng tiếp
 *     cho đến khi không trùng (giảm race condition ~99%).
 *  3. Không fallback bằng Date.now() khi lỗi → throw để route biết.
 *     Vì mã "KH1723456789" không đúng format sẽ phá vỡ UI/UX.
 *  4. Sanitize input để chống SQL injection qua tên bảng / cột / prefix.
 * 
 * ═══════════════════════════════════════════════════════════
 * GIỚI HẠN CÒN LẠI (0.1%):
 * ───────────────────────────────────────────────────────────
 *  Race condition thực sự 100% vẫn có thể xảy ra nếu 2 request đến
 *  trong cùng micro-giây giữa lúc check-duplicate và lúc INSERT.
 * 
 *  Để triệt để 100%, route cần bọc INSERT trong retry loop:
 *  
 *    let attempts = 0;
 *    while (attempts < 5) {
 *      try {
 *        const code = await generateNextCode('partners', 'code', 'KH', 3);
 *        const [r] = await pool.query('INSERT INTO partners (code, name) VALUES (?, ?)', [code, name]);
 *        break;
 *      } catch (e) {
 *        if (e.code === 'ER_DUP_ENTRY') { attempts++; continue; }
 *        throw e;
 *      }
 *    }
 *  
 *  → Sẽ áp dụng retry loop này khi sửa routes partners.js / items.js / orders.js.
 * 
 * ═══════════════════════════════════════════════════════════
 * 
 * @param {string} table       Tên bảng (VD: 'partners', 'items', 'orders')
 * @param {string} column      Tên cột mã (VD: 'code')
 * @param {string} prefix      Tiền tố (VD: 'KH', 'NCC', 'DH')
 * @param {number} padLength   Độ dài phần số (VD: 3 → KH001, 5 → DH00001)
 * @param {string} whereClause Điều kiện phụ (VD: "AND type = 'customer'")
 */
const generateNextCode = async (table, column, prefix, padLength = 3, whereClause = '') => {
  // ─── Sanitize inputs (chống SQL injection) ───
  const safeTable  = String(table).replace(/[^a-zA-Z0-9_]/g, '');
  const safeColumn = String(column).replace(/[^a-zA-Z0-9_]/g, '');
  const safePrefix = String(prefix).replace(/[^a-zA-Z0-9_-]/g, '');

  if (!safeTable || !safeColumn || !safePrefix) {
    throw new Error('generateNextCode: tham số không hợp lệ (table/column/prefix)');
  }

  try {
    // ─── Lấy số lớn nhất hiện có ───
    // VD prefix = 'KH', padLength = 3
    // → SUBSTRING(code, 3) cắt bỏ 'KH' → '001', '002', ...
    // → CAST AS UNSIGNED → 1, 2, ...
    // → MAX → số lớn nhất
    const sql = `
      SELECT MAX(CAST(SUBSTRING(${safeColumn}, ${safePrefix.length + 1}) AS UNSIGNED)) AS max_num
      FROM ${safeTable}
      WHERE ${safeColumn} LIKE ? ${whereClause}
    `;
    const [rows] = await pool.query(sql, [`${safePrefix}%`]);
    const maxNum = Number(rows[0]?.max_num) || 0;
    let nextNum = maxNum + 1;

    let code = `${safePrefix}${String(nextNum).padStart(padLength, '0')}`;

    // ─── Retry nội bộ (giảm race condition) ───
    // Nếu mã đã tồn tại (do race hoặc DB cũ có gap), tự tăng tiếp
    let safety = 0;
    while (safety < 100) {
      const [exist] = await pool.query(
        `SELECT 1 FROM ${safeTable} WHERE ${safeColumn} = ? LIMIT 1`,
        [code]
      );
      if (exist.length === 0) break;

      nextNum++;
      code = `${safePrefix}${String(nextNum).padStart(padLength, '0')}`;
      safety++;
    }

    if (safety >= 100) {
      throw new Error(`Không thể sinh mã ${safePrefix} — đã thử 100 lần vẫn trùng`);
    }

    return code;
  } catch (err) {
    console.error(`[generateNextCode] Lỗi sinh mã "${prefix}" cho "${table}":`, err.message);
    throw err; // KHÔNG fallback — route sẽ thấy lỗi và không INSERT với mã sai
  }
};

module.exports = { generateNextCode };
